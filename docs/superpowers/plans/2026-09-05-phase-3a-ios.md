**Spec:** [Shared Learning Contract](../specs/2026-09-05-shared-learning-contract-design.md), §3–8; [service plan](2026-09-05-phase-3a-service.md) cung cấp golden artifact.

```swift
struct SenseID: Hashable, Sendable, Codable {
    let rawValue: UUID
    init(rawValue: UUID) { self.rawValue = rawValue }
    init(from decoder: Decoder) throws {
        let box = try decoder.singleValueContainer()
        let text = try box.decode(String.self)
        guard let uuid = UUID(uuidString: text), uuid.uuidString.lowercased() == text else {
            throw DecodingError.dataCorruptedError(in: box, debugDescription: "Invalid sense ID")
        }
        rawValue = uuid
    }
    func encode(to encoder: Encoder) throws {
        var box = encoder.singleValueContainer()
        try box.encode(rawValue.uuidString.lowercased())
    }
}
```
